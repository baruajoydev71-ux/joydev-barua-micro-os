// Multi-Level Feedback Queue (MLFQ) Scheduler
// Implements 3 Priority Queues with Time-Slice Demotion and Anti-Starvation Aging

pub struct MlfqScheduler {
    pub queues: [TaskQueue; 3],       // Q0 (High), Q1 (Med), Q2 (Low)
    pub quantums: [u32; 3],          // 2 ticks, 4 ticks, 8 ticks
    pub tasks: [Option; MAX_TASKS],
    pub aging_threshold: u64,        // Priority boost after 10 ticks
    pub context_switches: u64,
}

impl MlfqScheduler {
    pub fn schedule_next(&mut self, current_time: u64) -> Option {
        // 1. Anti-Starvation: Priority Aging Boost
        if current_time - self.last_boost_time >= self.aging_threshold {
            self.boost_all_tasks();
            self.last_boost_time = current_time;
        }

        // 2. Select Highest Priority Runnable Task (Q0 -> Q1 -> Q2)
        for q_idx in 0..3 {
            if let Some(task_id) = self.queues[q_idx].pop_front() {
                return self.find_task_index(task_id);
            }
        }
        None
    }
}
